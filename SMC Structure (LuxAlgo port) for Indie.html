SMC Structure (LuxAlgo port) for Indie

Swing BOS/CHoCH are solid black lines and internal BOS/CHoCH are dashed lines, each from the pivot candle to the break candle with text in the middle. This version is untested. Paste it into the Exness Terminal indicator editor.

Based on "Smart Money Concepts" by LuxAlgo (CC BY-NC-SA 4.0, non-commercial).

# indie:lang_version = 5
# Port of "Smart Money Concepts" by LuxAlgo (CC BY-NC-SA 4.0, non-commercial)
# Swing (solid black) + Internal (dashed) BOS / CHoCH: line from pivot candle to break candle, text in the middle

import math
from indie import indicator, param, color, Var
from indie.drawings import LineSegment, LabelAbs, AbsolutePosition
from indie.algorithms import Highest, Lowest


def structure(high_s: float, low_s: float, time_s: float, close_0: float, close_1: float,
              hi: float, lo: float,
              leg: Var[int], ph: Var[float], pl: Var[float],
              ph_t: Var[float], pl_t: Var[float],
              ph_crossed: Var[bool], pl_crossed: Var[bool],
              trend: Var[int]) -> int:
    # event: 0 none, 1 bull BOS, 2 bull CHoCH, -1 bear BOS, -2 bear CHoCH
    ev: int = 0
    prev_leg: int = leg.get()
    if high_s > hi:
        leg.set(0)
    elif low_s < lo:
        leg.set(1)

    if leg.get() != prev_leg:
        if leg.get() == 1:
            pl.set(low_s)
            pl_t.set(time_s)
            pl_crossed.set(False)
        else:
            ph.set(high_s)
            ph_t.set(time_s)
            ph_crossed.set(False)

    if (not math.isnan(ph.get()) and not ph_crossed.get()
            and close_0 > ph.get() and close_1 <= ph.get()):
        if trend.get() == -1:
            ev = 2
        else:
            ev = 1
        ph_crossed.set(True)
        trend.set(1)

    if (not math.isnan(pl.get()) and not pl_crossed.get()
            and close_0 < pl.get() and close_1 >= pl.get()):
        if trend.get() == 1:
            ev = -2
        else:
            ev = -1
        pl_crossed.set(True)
        trend.set(-1)

    return ev


@indicator('SMC Structure (LuxAlgo port)', overlay_main_pane=True)
@param.int('swing_len', default=50, min=10, title='Swing Length')
@param.int('internal_len', default=5, min=2, title='Internal Length')
@param.bool('show_internal', default=True, title='Show Internal Structure')
def Main(self, swing_len: int, internal_len: int, show_internal: bool):
    hi_sw = Highest.new(self.high, swing_len)[0]
    lo_sw = Lowest.new(self.low, swing_len)[0]
    hi_in = Highest.new(self.high, internal_len)[0]
    lo_in = Lowest.new(self.low, internal_len)[0]

    s_leg = Var[int].new(0)
    s_ph = Var[float].new(math.nan)
    s_pl = Var[float].new(math.nan)
    s_pht = Var[float].new(math.nan)
    s_plt = Var[float].new(math.nan)
    s_phc = Var[bool].new(True)
    s_plc = Var[bool].new(True)
    s_trend = Var[int].new(0)

    i_leg = Var[int].new(0)
    i_ph = Var[float].new(math.nan)
    i_pl = Var[float].new(math.nan)
    i_pht = Var[float].new(math.nan)
    i_plt = Var[float].new(math.nan)
    i_phc = Var[bool].new(True)
    i_plc = Var[bool].new(True)
    i_trend = Var[int].new(0)

    s_ev: int = structure(self.high[swing_len], self.low[swing_len], self.time[swing_len],
                          self.close[0], self.close[1], hi_sw, lo_sw,
                          s_leg, s_ph, s_pl, s_pht, s_plt, s_phc, s_plc, s_trend)
    i_ev: int = structure(self.high[internal_len], self.low[internal_len], self.time[internal_len],
                          self.close[0], self.close[1], hi_in, lo_in,
                          i_leg, i_ph, i_pl, i_pht, i_plt, i_phc, i_plc, i_trend)

    now = self.time[0]

    # ---------- SWING (thin solid black line, text in the middle) ----------
    if s_ev != 0:
        t1: float = s_plt.get()
        lvl: float = s_pl.get()
        if s_ev > 0:
            t1 = s_pht.get()
            lvl = s_ph.get()
        txt: str = 'BOS'
        if s_ev == 2 or s_ev == -2:
            txt = 'CHoCH'
        self.chart.draw(LineSegment(AbsolutePosition(t1, lvl), AbsolutePosition(now, lvl), color=color.BLACK))
        self.chart.draw(LabelAbs(txt, AbsolutePosition((t1 + now) / 2.0, lvl)))

    # ---------- INTERNAL (dashed, max 40 dashes) ----------
    if show_internal and i_ev != 0:
        it1: float = i_plt.get()
        ilvl: float = i_pl.get()
        icol = color.RED(alpha=0.6)
        if i_ev > 0:
            it1 = i_pht.get()
            ilvl = i_ph.get()
            icol = color.GREEN(alpha=0.6)
        itxt: str = 'iBOS'
        if i_ev == 2 or i_ev == -2:
            itxt = 'iCHoCH'

        total: float = now - it1
        bar_len: float = self.time[0] - self.time[1]
        dash_len: float = bar_len
        if total / (2.0 * bar_len) > 40.0:
            dash_len = total / 80.0
        n: int = int(total / (2.0 * dash_len))
        if n > 40:
            n = 40
        for k in range(n):
            a: float = it1 + (2.0 * k) * dash_len
            b: float = a + dash_len
            self.chart.draw(LineSegment(AbsolutePosition(a, ilvl), AbsolutePosition(b, ilvl), color=icol))

        self.chart.draw(LabelAbs(itxt, AbsolutePosition((it1 + now) / 2.0, ilvl), bg_color=icol))
